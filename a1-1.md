## Example Automated Tests (Python + pytest)

Here are API-level tests, assuming an endpoint `POST /api/upload` that accepts `multipart/form-data` and returns the stored file's `id` and `filename`:

```python
import io

import pytest
import requests

BASE_URL = "http://localhost:8000"
UPLOAD_URL = f"{BASE_URL}/api/upload"
MAX_SIZE = 10 * 1024 * 1024  # 10 MB

# Minimal valid file headers ("magic bytes")
PNG_HEADER = b"\x89PNG\r\n\x1a\n"
PDF_HEADER = b"%PDF-1.4\n"


def login(username, password):
    resp = requests.post(f"{BASE_URL}/api/login",
                         json={"username": username, "password": password})
    return {"Authorization": f"Bearer {resp.json()['token']}"}


@pytest.fixture
def auth_headers():
    return login("testuser", "testpass")


def upload(headers, filename, content, content_type="application/octet-stream"):
    files = {"file": (filename, io.BytesIO(content), content_type)}
    return requests.post(UPLOAD_URL, files=files, headers=headers)


# --- Functional ---

def test_upload_valid_png(auth_headers):
    resp = upload(auth_headers, "image.png", PNG_HEADER + b"0" * 1024, "image/png")
    assert resp.status_code == 201
    assert "id" in resp.json()


def test_uploaded_file_can_be_downloaded_unchanged(auth_headers):
    content = PDF_HEADER + b"test content"
    resp = upload(auth_headers, "doc.pdf", content, "application/pdf")
    file_id = resp.json()["id"]

    download = requests.get(f"{BASE_URL}/api/files/{file_id}", headers=auth_headers)
    assert download.status_code == 200
    assert download.content == content


# --- File type validation ---

@pytest.mark.parametrize("filename", ["program.exe", "script.js", "archive.zip"])
def test_disallowed_file_types_rejected(auth_headers, filename):
    resp = upload(auth_headers, filename, b"some content")
    assert resp.status_code == 400


def test_executable_renamed_as_png_rejected(auth_headers):
    exe_content = b"MZ\x90\x00" + b"0" * 100  # Windows executable header
    resp = upload(auth_headers, "image.png", exe_content, "image/png")
    assert resp.status_code == 400


# --- File size validation ---

def test_empty_file_rejected(auth_headers):
    resp = upload(auth_headers, "empty.png", b"", "image/png")
    assert resp.status_code == 400


def test_file_at_max_size_accepted(auth_headers):
    content = PNG_HEADER + b"0" * (MAX_SIZE - len(PNG_HEADER))
    resp = upload(auth_headers, "max.png", content, "image/png")
    assert resp.status_code == 201


def test_file_over_max_size_rejected(auth_headers):
    content = PNG_HEADER + b"0" * (MAX_SIZE - len(PNG_HEADER) + 1)
    resp = upload(auth_headers, "too_big.png", content, "image/png")
    assert resp.status_code == 413


# --- Security ---

def test_upload_requires_authentication():
    resp = upload({}, "image.png", PNG_HEADER, "image/png")
    assert resp.status_code == 401


def test_path_traversal_filename_is_sanitized(auth_headers):
    resp = upload(auth_headers, "../../etc/passwd.png", PNG_HEADER, "image/png")
    assert resp.status_code in (201, 400)
    if resp.status_code == 201:
        stored_name = resp.json()["filename"]
        assert ".." not in stored_name
        assert "/" not in stored_name


def test_cannot_access_other_users_file(auth_headers):
    resp = upload(auth_headers, "private.pdf", PDF_HEADER, "application/pdf")
    file_id = resp.json()["id"]

    other_headers = login("otheruser", "otherpass")
    resp = requests.get(f"{BASE_URL}/api/files/{file_id}", headers=other_headers)
    assert resp.status_code in (403, 404)
```

## Tips

- **Generate test files programmatically** (as above) rather than committing large binary fixtures to your repo.
- **Test both client-side and server-side validation.** Client-side checks can be bypassed, so the server must enforce every rule.
- **Use the EICAR test file** for antivirus checks; it is harmless but detected as a virus by scanners.
- **Clean up** uploaded test files after each run so tests stay independent.
